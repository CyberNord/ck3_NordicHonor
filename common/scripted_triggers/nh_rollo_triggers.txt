# Rollo's scripted event war
nh_rollo_adventurer_eligible_trigger = {
	has_dlc_feature = roads_to_power
	is_alive = yes
	this = character:242
	religion = religion:germanic_religion
	faith = { has_doctrine_parameter = unreformed }
	is_landed = no
	government_has_flag = government_is_landless_adventurer
	top_liege = this
	exists = domicile.domicile_location
	is_capable_adult = yes
	is_imprisoned = no
	is_at_war = no
	NOT = { exists = involved_activity }
	OR = {
		is_ai = no
		AND = {
			is_ai = yes
			current_date >= 905.1.1
			prestige_level >= 2
		}
	}
}

nh_rollo_mission_available_trigger = {
	nh_rollo_adventurer_eligible_trigger = yes
	NOT = { has_character_flag = nh_rollo_mission_started }
	NOT = {
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:nh_rollos_adventure_decision
		}
	}
}

# Ownership is resolved at selection and again immediately before declaration.
nh_rollo_adventure_target_valid_trigger = {
	nh_rollo_adventurer_eligible_trigger = yes
	NOT = { has_character_flag = nh_rollo_mission_war_started }
	OR = {
		is_ai = no
		$TITLE$ = { this = title:d_normandy }
	}
	exists = $TITLE$
	exists = $COUNTY$.holder
	exists = $COUNTY$.holder.top_liege
	$COUNTY$.holder.top_liege = {
		is_alive = yes
		is_landed = yes
		NOT = { this = character:242 }
	}
	$TITLE$ = {
		any_in_de_jure_hierarchy = {
			tier = tier_county
			exists = holder
			holder.top_liege = $COUNTY$.holder.top_liege
		}
	}
}

nh_rollo_any_adventure_target_valid_trigger = {
	OR = {
		nh_rollo_adventure_target_valid_trigger = { COUNTY = title:c_rouen TITLE = title:d_normandy }
		nh_rollo_adventure_target_valid_trigger = { COUNTY = title:c_cagliari TITLE = title:d_sardinia }
		nh_rollo_adventure_target_valid_trigger = { COUNTY = title:c_middlesex TITLE = title:d_bedford }
	}
}

nh_rollo_normandy_offer_valid_trigger = {
	is_alive = yes
	this = character:242
	is_landed = yes
	is_at_war = no
	has_title = title:d_normandy
	is_confederation_member = no
	is_tributary = no
	exists = title:k_france.holder
	exists = title:k_france.holder.top_liege
	title:k_france.holder.top_liege = {
		is_alive = yes
		is_landed = yes
		NOT = { government_has_flag = cannot_be_vassal_or_liege }
		NOT = { government_has_flag = government_is_herder }
	}
	highest_held_title_tier < title:k_france.holder.top_liege.highest_held_title_tier
	NOT = { government_has_flag = cannot_be_vassal_or_liege }
	NOT = { government_has_flag = government_is_landless_adventurer }
	NOT = { government_has_flag = government_is_herder }
	OR = {
		top_liege = this
		is_vassal_of = title:k_france.holder.top_liege
	}
}

nh_rollo_normandy_acceptance_valid_trigger = {
	nh_rollo_normandy_offer_valid_trigger = yes
	exists = scope:francish_king
	scope:francish_king = { is_alive = yes }
	title:k_france.holder.top_liege = scope:francish_king
	exists = capital_province
	can_marry_common_trigger = yes
}
