# Shared safety envelope for the four existing acceptance options. 
nh_rollo_accept_normandy_effect = {
	save_scope_value_as = { name = nh_story_conversion_succeeded value = no }
	if = {
		limit = {
			nh_rollo_normandy_acceptance_valid_trigger = yes
			OR = {
				AND = {
					always = $HISTORICAL$
					scope:francish_king = character:90104
					exists = character:167830
				}
				AND = {
					always = $ALTERNATE$
					NOT = { scope:francish_king = character:90104 }
				}
			}
		}
		save_scope_as = actor
		$BRIDE_EFFECT$ = yes
		if = {
			limit = {
				exists = scope:norman_spouse
				scope:norman_spouse = { is_alive = yes }
				could_marry_character_trigger = { CHARACTER = scope:norman_spouse }
			}
			if = {
				limit = { always = $CONVERT$ }
				nh_story_convert_to_frankish_rite_effect = yes
			}
			if = {
				limit = {
					OR = {
						always = $KEEP_FAITH$
						scope:nh_story_conversion_succeeded = yes
					}
				}
				if = {
					limit = { exists = primary_spouse }
					divorce = primary_spouse
				}
				marry = scope:norman_spouse
				if = {
					limit = { is_spouse_of = scope:norman_spouse }
					nh_swear_fealty_effect = yes
					if = {
						limit = { is_vassal_of = scope:francish_king }
						$CONTRACT_EFFECT$ = yes
						dynasty = { add_dynasty_prestige = 500 }
						if = {
							limit = { always = $CONVERT$ }
							scope:francish_king = { trigger_event = nh.9000 }
						}
					}
				}
			}
		}
	}
}

nh_swear_fealty_effect = {
	# Already a direct vassal: keep the existing relationship and contract.
	if = {
		limit = {
			nh_rollo_normandy_acceptance_valid_trigger = yes
			NOT = { is_vassal_of = scope:francish_king }
		}
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = change
		}

		scope:actor = {
			change_liege = {
				liege = scope:francish_king
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change

		if = {
			limit = { is_vassal_of = scope:francish_king }
			add_opinion = {
				modifier = became_vassal
				target = scope:francish_king
				opinion = 10
			}
			# Remove the old imprisonment reason only after actual submission.
			consume_all_criminal_reasons_effect = {
				LIEGE = scope:francish_king
				CRIMINAL = scope:actor
			}
		}
	}
}

nh_story_convert_to_rite_effect = {
	save_scope_value_as = { name = nh_story_conversion_succeeded value = no }
	if = {
		limit = {
			is_alive = yes
			exists = faith
			exists = $RITE$
			$RITE$ = { is_rite_enabled = yes can_convert_to_rite = yes }
		}
		$RITE$ = { save_scope_as = nh_story_target_rite }
		if = {
			limit = {
				NOT = { faith = scope:nh_story_target_rite.faith }
				is_adult = yes
				NOT = { faith.religious_head = this }
				pam_is_antipope_trigger = no
				is_in_ongoing_great_holy_war = no
				NOR = { has_title = title:k_papal_state has_title = title:k_orthodox }
				scope:nh_story_target_rite.faith = {
					NOT = { has_variable = block_conversion_till_decision_taken }
					NOT = { has_variable = block_conversion_till_nebulous_circumstances }
					main_rite = { is_rite_enabled = yes can_convert_to_rite = yes }
				}
			}
			add_known_rite = scope:nh_story_target_rite.faith.main_rite
			add_known_rite = scope:nh_story_target_rite
			remove_character_flag = faith_conversion_recently_converted
			set_character_faith_with_conversion = scope:nh_story_target_rite.faith
		}
		if = {
			limit = { faith = scope:nh_story_target_rite.faith }
			if = {
				limit = { NOT = { rite = scope:nh_story_target_rite } }
				set_character_rite = scope:nh_story_target_rite
			}
			if = {
				limit = { faith = scope:nh_story_target_rite.faith rite = scope:nh_story_target_rite }
				if = {
					limit = { exists = capital_county capital_county.faith = faith }
					capital_county = {
						if = {
							limit = { NOT = { rite = scope:nh_story_target_rite } }
							set_county_rite = scope:nh_story_target_rite
						}
					}
				}
				save_scope_value_as = { name = nh_story_conversion_succeeded value = yes }
			}
		}
	}
}

# v1.2 - Forced Conversion Effect
# County scope; caller supplies scope:converter. Named scopes must survive the
# province/character iterators and delayed outcome events without relying on PREV.
nh_convert_county_forced_effect = {

	title_province.faith = {
		save_scope_as = nh_converted_county_old_faith
	}
	
	# Convert county and all holdings
	set_county_rite = scope:converter.rite
	
	if = {
		limit = { scope:converter.rite = rite:odinstru_pagan }
		change_development_level = {
			subtract = { value = development_level multiply = 0.15 }
		}
	}
	else = {
		change_development_level = {
			subtract = { value = development_level multiply = 0.20 }
		}
	}
	
	add_county_modifier = {
		modifier = nh_forced_conversion
		years = 10
	}
	
	scope:converter = {
		# Dread is an immediate effect, not a persistent modifier field.
		add_dread = minor_dread_gain
		add_character_modifier = {
			modifier = nh_religious_oppressor
			years = 10
		}
	}
	
	# Imprison heretics
	every_county_province = {
		every_character_in_location = {
			limit = {
				NOT = { faith = scope:converter.faith }
				NOT = { this = scope:converter }
				is_imprisoned = no
				is_ruler = no
			}
			save_scope_as = nh_conversion_prisoner
			scope:converter = {
				imprison = {
					target = scope:nh_conversion_prisoner
					type = house_arrest
				}
			}
		}
	}
	
	# Odinstru bonus
	if = {
		limit = { scope:converter.rite = rite:odinstru_pagan }
		scope:converter = {
			add_dread = minor_dread_gain
			add_character_modifier = {
				modifier = nh_champion_of_faith
				years = 2
			}
			every_vassal = {
				limit = { faith = scope:converter.faith }
				add_opinion = {
					target = scope:converter
					modifier = approved_conversion_opinion
				}
			}
		}
	}
	
	scope:converter = {
		every_neighboring_and_across_water_realm_same_rank_owner = {
			limit = { faith = scope:nh_converted_county_old_faith }
			add_opinion = {
				target = scope:converter
				modifier = forced_conversion_opinion
			}
		}
	}
}
