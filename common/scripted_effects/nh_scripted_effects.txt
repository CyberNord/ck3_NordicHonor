# v1.2 - Spawn elite warriors based on realm size
nh_spawn_elite_warriors_effect = {
	# Small realm (< 5 counties)
	if = {
		limit = { realm_size < 5 }
		spawn_army = {
			men_at_arms = {
				type = vigmen
				stacks = 1
			}
			men_at_arms = {
				type = varangian_veterans
				stacks = 1
			}
			men_at_arms = {
				type = bondi
				stacks = 23
			}
			location = root.capital_province
			origin = root.capital_province
			inheritable = yes
			uses_supply = yes
			name = event_troop_elite_warriors_name
		}
	}
	# Medium realm (5-10 counties)
	else_if = {
		limit = {
			realm_size >= 5
			realm_size <= 10
		}
		spawn_army = {
			men_at_arms = {
				type = vigmen
				stacks = 2
			}
			men_at_arms = {
				type = varangian_veterans
				stacks = 2
			}
			men_at_arms = {
				type = huscarl
				stacks = 1
			}
			men_at_arms = {
				type = bondi
				stacks = 20
			}
			location = root.capital_province
			origin = root.capital_province
			inheritable = yes
			uses_supply = yes
			name = event_troop_elite_warriors_name
		}
	}
	# Large realm (> 10 counties)
	else = {
		spawn_army = {
			men_at_arms = {
				type = vigmen
				stacks = 5
			}
			men_at_arms = {
				type = varangian_veterans
				stacks = 3
			}
			men_at_arms = {
				type = huscarl
				stacks = 2
			}
			men_at_arms = {
				type = bondi
				stacks = 15
			}
			location = root.capital_province
			origin = root.capital_province
			inheritable = yes
			uses_supply = yes
			name = event_troop_elite_warriors_name
		}
	}
}

# Scripted story choices may cross faith boundaries, unlike the rite-view conversion.
# Convert to the target faith first (including willing vassals/capital), then select
# the promised rite explicitly. Do not use a history effect: conversion hooks matter.
nh_story_convert_to_rite_effect = {
	if = {
		limit = { NOT = { faith = $RITE$.faith } }
		set_character_faith_with_conversion = $RITE$.faith
	}
	set_character_rite = $RITE$
	if = {
		limit = { exists = capital_county capital_county.faith = faith }
		capital_county = { set_county_rite = $RITE$ }
	}
}

# v1.2 - Forced Conversion Effect
# County scope; caller supplies scope:converter. Named scopes must survive the
# province/character iterators and delayed outcome events without relying on PREV.
nh_convert_county_forced_effect = {

	title_province.faith = {
		save_scope_as = nh_converted_county_old_faith
	}
	
	# Convert county and all holdings
	set_county_rite = scope:converter.rite
	
	if = {
		limit = { scope:converter.rite = rite:odinstru_pagan }
		change_development_level = {
			subtract = { value = development_level multiply = 0.15 }
		}
	}
	else = {
		change_development_level = {
			subtract = { value = development_level multiply = 0.20 }
		}
	}
	
	add_county_modifier = {
		modifier = nh_forced_conversion
		years = 10
	}
	
	scope:converter = {
		# Dread is an immediate effect, not a persistent modifier field.
		add_dread = minor_dread_gain
		add_character_modifier = {
			modifier = nh_religious_oppressor
			years = 10
		}
	}
	
	# Imprison heretics
	every_county_province = {
		every_character_in_location = {
			limit = {
				NOT = { faith = scope:converter.faith }
				NOT = { this = scope:converter }
				is_imprisoned = no
				is_ruler = no
			}
			save_scope_as = nh_conversion_prisoner
			scope:converter = {
				imprison = {
					target = scope:nh_conversion_prisoner
					type = house_arrest
				}
			}
		}
	}
	
	# Odinstru bonus
	if = {
		limit = { scope:converter.rite = rite:odinstru_pagan }
		scope:converter = {
			add_dread = minor_dread_gain
			add_character_modifier = {
				modifier = nh_champion_of_faith
				years = 2
			}
			every_vassal = {
				limit = { faith = scope:converter.faith }
				add_opinion = {
					target = scope:converter
					modifier = approved_conversion_opinion
				}
			}
		}
	}
	
	scope:converter = {
		every_neighboring_and_across_water_realm_same_rank_owner = {
			limit = { faith = scope:nh_converted_county_old_faith }
			add_opinion = {
				target = scope:converter
				modifier = forced_conversion_opinion
			}
		}
	}
}
