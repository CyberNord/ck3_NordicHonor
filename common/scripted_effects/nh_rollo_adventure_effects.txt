# Rollo's one-shot mission; AI and decision entry share the same safety gates.
nh_rollo_begin_adventure_effect = {
	if = {
		limit = {
			nh_rollo_mission_available_trigger = yes
			nh_rollo_any_adventure_target_valid_trigger = yes
		}
		add_character_flag = nh_rollo_mission_started
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:nh_rollos_adventure_decision
		}
		save_scope_as = actor
		trigger_event = nh.1011
	}
}

# The pending references identify this declaration in the native war-start callback.
nh_rollo_start_adventure_war_effect = {
	if = {
		limit = {
			has_character_flag = nh_rollo_mission_started
			nh_rollo_adventure_target_valid_trigger = { COUNTY = $COUNTY$ TITLE = $TITLE$ }
		}
		add_character_flag = nh_rollo_mission_war_started
		set_variable = { name = nh_rollo_pending_target value = $TITLE$ days = 1 }
		set_variable = { name = nh_rollo_pending_defender value = $COUNTY$.holder.top_liege days = 1 }
		start_war = {
			cb = ep3_laamp_adventurer_duchy_invasion
			target = $COUNTY$.holder.top_liege
			target_title = $TITLE$
		}
	}
}

# Called one day after native war resolution, after camp and title cleanup.
nh_rollo_finish_adventure_victory_effect = {
	if = {
		limit = {
			this = character:242
			is_alive = yes
			has_character_flag = nh_rollo_victory_pending
			NOT = { has_character_flag = nh_rollo_victory_resolved }
		}
		add_character_flag = nh_rollo_victory_resolved
		remove_character_flag = nh_rollo_victory_pending
		if = {
			limit = {
				is_landed = yes
				exists = var:nh_rollo_victory_target
				var:nh_rollo_victory_target = {
					any_in_de_jure_hierarchy = {
						tier = tier_county
						holder = root
					}
				}
			}
			add_gold = var:nh_rollo_victory_bounty
			if = {
				limit = { has_variable = va_cb_tally }
				change_variable = { name = va_cb_tally add = 1 }
			}
			else = { set_variable = { name = va_cb_tally value = 2 } }
			if = {
				limit = {
					is_ai = yes
					var:nh_rollo_victory_target = title:d_normandy
					has_title = title:d_normandy
					exists = capital_province
					NOT = { has_character_flag = nh_rollo_ai_normandy_rewarded }
				}
				add_character_flag = nh_rollo_ai_normandy_rewarded
				hidden_effect = {
					add_character_modifier = nh_rollo_ai_normandy_modifier
					spawn_army = {
						name = event_troop_rollo_name
						men_at_arms = { type = varangian_veterans stacks = 10 }
						men_at_arms = { type = vigmen stacks = 5 }
						men_at_arms = { type = bondi stacks = 5 }
						location = capital_province
						origin = capital_province
						inheritable = no
					}
				}
			}
		}
		remove_variable = nh_rollo_victory_target
		remove_variable = nh_rollo_victory_bounty
	}
}
