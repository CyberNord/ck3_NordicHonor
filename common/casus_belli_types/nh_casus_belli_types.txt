# Nordic Honor - Religious Conquest Casus Belli
# Restricted to the Óðinnstrú rite.

nh_holy_conquest_cb = {
	icon = county_conquest_cb
	group = conquest

	should_show_war_goal_subview = no

	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150

	allowed_for_character = {
		# Must be Óðinnstrú
		rite = rite:odinstru_pagan
		
		# Basic conquest CB requirements
		can_use_conquest_cbs_trigger = yes
		
		NOT = {
			government_has_flag = government_is_landless_adventurer
		}
	}

	allowed_for_character_display_regardless = {
		# Minimum devotion blocks declaration without hiding the CB.
		piety_level >= 2

		# Struggle constraint
		custom_tooltip = {
			text = struggle_involvement_cb_check_failure
			struggle_blocks_invasion_conquest_cb_trigger = yes
		}
	}

	allowed_against_character = {
		nh_holy_raid_access_trigger = yes
	}

	target_titles = none


	ai_score_mult = {
		value = viking_conquest_ai_score_value

		# House relations
		add = house_relation_ai_score_value
		
		# Struggle constraints
		multiply = struggle_wars_prioritise_struggle_targets_value
		
		# Boost for zealous characters
		if = {
			limit = { scope:attacker = { has_trait = zealous } }
			multiply = 1.5
		}
	}

	valid_to_start = {
		nh_holy_raid_access_trigger = yes
	}

	valid_to_start_display_regardless = {
		custom_tooltip = {
			text = nh_holy_raid_cooldown_tt
			nh_holy_raid_cooldown_ready_trigger = yes
		}
	}

	should_invalidate = {
		NOT = { nh_holy_raid_access_trigger = yes }
	}

	on_invalidated_desc = { desc = msg_invalidate_war_title }
	on_invalidated = {}

	cost = {
		piety = {
			value = 500
			desc = CB_HOLY_WAR_COST
		}
		
		prestige = {
			value = 100
			desc = CB_CONQUEST_COST
			
			add = {
				value = 5
				multiply = scope:attacker.realm_size
				desc = CB_REALM_SIZE_COST
				max = 150
			}
			
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = nh_holy_conquest_cb_victory_desc_attacker
			}
			desc = nh_holy_conquest_cb_victory_desc
		}
	}

	on_victory = {
		scope:attacker = { 
			show_pow_release_message_effect = yes 
			accolade_attacker_war_end_glory_gain_med_effect = yes
		}
		
		# Legitimacy gain
		add_legitimacy_attacker_victory_effect = yes

		# Prestige for attacker
		scope:attacker = {
			add_prestige_experience = medium_prestige_value
		}

		# Allied contribution rewards
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = yes
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		# Truce
		add_truce_attacker_victory_effect = yes

		# FP1: stele memory
		scope:attacker = { fp1_remember_recent_conquest_victory_effect = yes }

		nh_holy_raid_full_victory_effect = {
			WINNER = scope:attacker
			LOSER = scope:defender
			FERVOR_LOSS = -5
		}
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = nh_holy_conquest_cb_white_peace_desc_defender
			}
			desc = nh_holy_conquest_cb_white_peace_desc
		}
	}

	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		
		# Minor prestige loss for attacker
		scope:attacker = {
			add_prestige = {
				value = medium_prestige_value
				multiply = -0.5
			}
			stress_impact = {
				ambitious = medium_stress_impact_gain
				zealous = medium_stress_impact_gain
			}
		}

		# Allied contribution rewards
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = yes
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		scope:defender = {
			stress_impact = {
				arrogant = medium_stress_impact_gain
			}
		}

		add_truce_white_peace_effect = yes

		nh_holy_raid_white_peace_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = nh_holy_conquest_cb_defeat_desc_defender
			}
			triggered_desc = {
				trigger = {
					scope:attacker = {
						is_local_player = yes
						has_targeting_faction = yes
					}
				}
				desc = nh_holy_conquest_cb_defeat_desc_attacker
			}
			desc = nh_holy_conquest_cb_defeat_desc
		}
	}

	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }

		# Legitimacy for the winner
		add_legitimacy_attacker_defeat_effect = yes

		# Influence for an eligible victorious defender
		add_influence_attacker_defeat_effect = yes

		scope:defender = {
			accolade_defender_war_end_glory_gain_low_effect = yes
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = major_prestige_value
			}
		}

		# Heavy penalties for attacker
		scope:attacker = {
			add_prestige = major_prestige_loss
			add_piety = {
				value = minor_piety_value
				multiply = -1
			}
		}

		# Allied contribution rewards
		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = yes
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		add_truce_attacker_defeat_effect = yes

		scope:attacker = {
			save_temporary_scope_as = loser
		}
		on_lost_aggression_war_discontent_loss = yes

		nh_holy_raid_full_victory_effect = {
			WINNER = scope:defender
			LOSER = scope:attacker
			FERVOR_LOSS = -25
		}
	}

	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "NH_HOLY_CONQUEST_WAR_NAME"
	war_name_base = "NH_HOLY_CONQUEST_WAR_NAME_BASE"
	cb_name = "NH_HOLY_CONQUEST_CB_NAME"

	interface_priority = 90


	attacker_wargoal_percentage = 0.8

	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}
