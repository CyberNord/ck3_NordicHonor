# Called from code after history generation
# Empty scope
on_game_start = {
	events = {
	}
	on_actions = {
		nh_on_game_start_setup
		nh_on_game_start_rollo
		nh_on_game_start_ivar_perks
	}
}

# Historical start troops, nickname, trait and county cultures.
nh_on_game_start_setup = {
	effect = {
		if = {
			limit = {
				game_start_date <= 993.1.1
			}
			character:180257 = {
				spawn_army = {
					name = event_troop_default_name
					men_at_arms = {
						type = huscarl
						stacks = 5
					}
					location = province:4
					origin = province:4
					inheritable = no
				}
			}
			character:168609 = {
				remove_nickname = yes
				give_nickname = nick_nh_wise
			}
			character:6811 = {
				add_trait = einherjar
				spawn_army = {
					name = event_troop_default_name
					men_at_arms = {
						type = huscarl
						stacks = 10
					}
					location = province:2157
					origin = province:2157
					inheritable = yes
				}
			}
			title:c_inner_hebrides = {
				set_county_culture = culture:norse
			}
			title:c_east_riding = {
				set_county_culture = culture:norse
			}
			title:c_north_riding = {
				set_county_culture = culture:norse
			}
		}
	}
}


nh_on_game_start_rollo = {

			}
		 }
	}
}

# Historical Ivar: fixed 867 start perks, guarded against repeated initialization.
nh_on_game_start_ivar_perks = {
	trigger = {
		game_start_date = 867.1.1
		current_date = 867.1.1
		exists = character:163111
	}
	effect = {
		character:163111 = {
			if = {
				limit = {
					is_alive = yes
					NOT = { has_character_flag = nh_ivar_torturer_start_initialized }
				}
				if = {
					limit = { NOT = { has_perk = dark_insights_perk } }
					add_perk = dark_insights_perk
				}
				if = {
					limit = { NOT = { has_perk = divine_retribution_perk } }
					add_perk = divine_retribution_perk
				}
				if = {
					limit = { NOT = { has_perk = dreadful_perk } }
					add_perk = dreadful_perk
				}
				if = {
					limit = { NOT = { has_perk = thriving_in_chaos_perk } }
					add_perk = thriving_in_chaos_perk
				}
				if = {
					limit = { NOT = { has_perk = malice_implicit_perk } }
					add_perk = malice_implicit_perk
				}
				if = {
					limit = { NOT = { has_perk = torturer_perk } }
					add_perk = torturer_perk
				}
				# The trait is added only if the capstone grant did not already apply it.
				if = {
					limit = {
						has_perk = torturer_perk
						NOT = { has_trait = torturer }
					}
					add_trait = torturer
				}
				if = {
					limit = {
						has_perk = dark_insights_perk
						has_perk = divine_retribution_perk
						has_perk = dreadful_perk
						has_perk = thriving_in_chaos_perk
						has_perk = malice_implicit_perk
						has_perk = torturer_perk
						has_trait = torturer
					}
					add_character_flag = nh_ivar_torturer_start_initialized
				}
			}
		}
	}
}
